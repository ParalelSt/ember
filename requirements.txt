# Python dependencies for player.py — the YouTube Music / yt-dlp bridge
# the Next API spawns as a subprocess. Touched by the API every time
# someone searches, plays, or downloads.
#
# After pulling a new version of this file, install/upgrade with:
#
#   .venv/bin/pip install --upgrade -r requirements.txt
#
# On a fresh checkout with no venv yet:
#
#   python3 -m venv .venv
#   .venv/bin/pip install -r requirements.txt
#
# Then restart the Next server (./start-static.sh or npm run dev) so
# the next subprocess spawn picks up the new Python packages.
#
# ytmusicapi is pinned to >=1.12.0 because earlier versions crash with
# `KeyError: 'musicShelfRenderer'` on certain YouTube Music queries —
# the upstream guard was added in 1.12.0. yt-dlp stays unpinned so we
# can pull the latest whenever YouTube changes their backend.

ytmusicapi>=1.12.0
yt-dlp

# --- Generated guitar tabs (transcribe.py) -----------------------------------
# Basic Pitch is installed without its declared deps: on macOS + Python 3.13 it
# asks for tensorflow-macos, which has no wheel. The ONNX runtime below is the
# backend it falls back to. Install order matters:
#
#   .venv/bin/pip install 'setuptools<80'
#   .venv/bin/pip install --no-deps basic-pitch
#   .venv/bin/pip install onnxruntime librosa pretty_midi 'resampy<0.4.3' mir_eval scikit-learn typing-extensions
#
# setuptools<80 because basic-pitch's resampy still imports pkg_resources.
#
# Optional, ~2GB with torch, but it is what makes the result a GUITAR tab
# rather than a transcription of the whole mix (vocals, bass, drums included):
#
#   .venv/bin/pip install demucs
#
# Without it transcribe.py falls back to the full mix and says so on stderr.
